    78→function getInitials(name: string): string {
    79→  return name
    80→    .trim()
    81→    .split(/\s+/)
    82→    .slice(0, 2)
    83→    .map((w) => w[0])
    84→    .join('')
    85→    .toUpperCase();
    86→}
    87→
    88→function relativeTime(dateStr: string): string {
    89→  const now = Date.now();
    90→  const date = new Date(dateStr).getTime();
    91→  const diffMs = now - date;
    92→  const diffSec = Math.floor(diffMs / 1000);
    93→  const diffMin = Math.floor(diffSec / 60);
    94→  const diffHour = Math.floor(diffMin / 60);
    95→  const diffDay = Math.floor(diffHour / 24);
    96→  const diffWeek = Math.floor(diffDay / 7);
    97→
    98→  if (diffSec < 60) return 'همین الان';
    99→  if (diffMin < 60) return formatNumber(diffMin) + ' دقیقه پیش';
   100→  if (diffHour < 24) return formatNumber(diffHour) + ' ساعت پیش';
   101→  if (diffDay < 7) return formatNumber(diffDay) + ' روز پیش';
   102→  if (diffWeek < 4) return formatNumber(diffWeek) + ' هفته پیش';
   103→
   104→  return new Intl.DateTimeFormat('fa-IR', {
   105→    month: 'short',
   106→    day: 'numeric',
   107→  }).format(new Date(dateStr));
   108→}
   109→
   110→function formatMessageTime(dateStr: string): string {
   111→  return new Intl.DateTimeFormat('fa-IR', {
   112→    hour: '2-digit',
   113→    minute: '2-digit',
   114→  }).format(new Date(dateStr));
   115→}
   116→
   117→function getOtherParticipant(
   118→  conversation: Conversation,
   119→  currentUserId: string | undefined,
   120→): Participant {
   121→  const other = conversation.participants.find(
   122→    (p) => p.id !== currentUserId,
   123→  );
   124→  return other || conversation.participants[0];
   125→}
   126→
   127→function truncate(str: string, len: number): string {
   128→  if (str.length <= len) return str;
   129→  return str.slice(0, len) + '…';
   130→}

   136→export default function MessagesClient() {
   137→  const { user } = useAuth();
   138→
   139→  /* ---- State ---- */
   140→  const [conversations, setConversations] = useState<Conversation[]>([]);
   141→  const [conversationsLoading, setConversationsLoading] = useState(true);
   142→  const [searchQuery, setSearchQuery] = useState('');
   143→
   144→  const [activeConversationId, setActiveConversationId] = useState<
   145→    string | null
   146→  >(null);
   147→  const [showChat, setShowChat] = useState(false); // mobile toggle
   148→
   149→  const [messages, setMessages] = useState<Message[]>([]);
   150→  const [messagesLoading, setMessagesLoading] = useState(false);
   151→  const [messagesMeta, setMessagesMeta] = useState<MessagesMeta | null>(null);
   152→  const [messageInput, setMessageInput] = useState('');
   153→  const [sendingMessage, setSendingMessage] = useState(false);
   154→
   155→  // New conversation dialog
   156→  const [newConvOpen, setNewConvOpen] = useState(false);
   157→  const [newConvParticipantId, setNewConvParticipantId] = useState('');
   158→  const [newConvProjectId, setNewConvProjectId] = useState('');
   159→  const [creatingConv, setCreatingConv] = useState(false);
   160→
   161→  const messagesEndRef = useRef<HTMLDivElement>(null);
   162→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   163→  const lastFetchedMessageId = useRef<string | null>(null);
   164→
   165→  const activeConversation = conversations.find(
   166→    (c) => c.id === activeConversationId,
   167→  );
   168→
   169→  /* ---- Fetch conversations ---- */
   170→  const fetchConversations = useCallback(async () => {
   171→    try {
   172→      const res = await fetch('/api/v1/conversations');
   173→      if (res.ok) {
   174→        const json = await res.json();
   175→        setConversations(json.data || []);
   176→      }
   177→    } catch {
   178→      // silent
   179→    } finally {
   180→      setConversationsLoading(false);
   181→    }
   182→  }, []);
   183→
   184→  /* ---- Fetch messages ---- */
   185→  const fetchMessages = useCallback(
   186→    async (conversationId: string, sinceId?: string) => {
   187→      try {
   188→        let url = `/api/v1/conversations/${conversationId}/messages?page=1&limit=50`;
   189→        if (sinceId) {
   190→          url += `&after=${sinceId}`;
   191→        }
   192→        const res = await fetch(url);
   193→        if (res.ok) {
   194→          const json = await res.json();
   195→          const newMessages: Message[] = json.data || [];
   196→          const meta: MessagesMeta | null = json.meta || null;
   197→
   198→          if (sinceId) {
   199→            // Only append new messages
   200→            setMessages((prev) => {
   201→              const existingIds = new Set(prev.map((m) => m.id));
   202→              const unique = newMessages.filter(
   203→                (m: Message) => !existingIds.has(m.id),
   204→              );
   205→              return [...prev, ...unique];
   206→            });
   207→          } else {
   208→            setMessages(newMessages);
   209→            setMessagesMeta(meta);
   210→          }
   211→
   212→          // Track last message ID for polling
   213→          if (newMessages.length > 0) {
   214→            lastFetchedMessageId.current =
   215→              newMessages[newMessages.length - 1].id;
   216→          }
   217→
   218→          return newMessages;
   219→        }
   220→      } catch {
   221→        // silent
   222→      }
   223→      return [];
   224→    },
   225→    [],
   226→  );
   227→
   228→  /* ---- Open conversation ---- */
   229→  const openConversation = useCallback(
   230→    async (conversationId: string) => {
   231→      setActiveConversationId(conversationId);
   232→      setMessages([]);
   233→      setMessagesLoading(true);
   234→      lastFetchedMessageId.current = null;
   235→      setShowChat(true);
   236→
   237→      await fetchMessages(conversationId);
   238→      setMessagesLoading(false);
   239→    },
   240→    [fetchMessages],
   241→  );
   242→
   243→  /* ---- Send message ---- */
   244→  const sendMessage = useCallback(async () => {
   245→    if (!activeConversationId || !messageInput.trim() || sendingMessage) return;
   246→
   247→    setSendingMessage(true);
   248→    try {
   249→      const res = await fetch(
   250→        `/api/v1/conversations/${activeConversationId}/messages`,
   251→        {
   252→          method: 'POST',
   253→          headers: { 'Content-Type': 'application/json' },
   254→          body: JSON.stringify({
   255→            content: messageInput.trim(),
   256→            type: 'TEXT',
   257→          }),
   258→        },
   259→      );
   260→      if (res.ok) {
   261→        const json = await res.json();
   262→        const newMsg: Message = json.data;
   263→        setMessages((prev) => [...prev, newMsg]);
   264→        lastFetchedMessageId.current = newMsg.id;
   265→        setMessageInput('');
   266→
   267→        // Reset textarea height
   268→        if (textareaRef.current) {
   269→          textareaRef.current.style.height = 'auto';
   270→        }
   271→
   272→        // Update conversation's last message optimistically
   273→        setConversations((prev) =>
   274→          prev.map((c) =>
   275→            c.id === activeConversationId
   276→              ? {
   277→                  ...c,
   278→                  lastMessage: {
   279→                    id: newMsg.id,
   280→                    content: newMsg.content,
   281→                    type: newMsg.type,
   282→                    createdAt: newMsg.createdAt,
   283→                    senderId: newMsg.senderId,
   284→                  },
   285→                  updatedAt: newMsg.createdAt,
   286→                }
   287→              : c,
   288→          ),
   289→        );
   290→      } else {
   291→        const json = await res.json().catch(() => null);
   292→        toast.error(
   293→          json?.error?.message || 'خطا در ارسال پیام',
   294→        );
   295→      }
   296→    } catch {
   297→      toast.error('خطا در ارسال پیام');
   298→    } finally {
   299→      setSendingMessage(false);
   300→    }
   301→  }, [activeConversationId, messageInput, sendingMessage]);
   302→
   303→  /* ---- Create new conversation ---- */
   304→  const createConversation = useCallback(async () => {
   305→    if (!newConvParticipantId.trim() || creatingConv) return;
   306→
   307→    setCreatingConv(true);
   308→    try {
   309→      const body: { participantId: string; projectId?: string } = {
   310→        participantId: newConvParticipantId.trim(),
   311→      };
   312→      if (newConvProjectId.trim()) {
   313→        body.projectId = newConvProjectId.trim();
   314→      }
   315→
   316→      const res = await fetch('/api/v1/conversations', {
   317→        method: 'POST',
   318→        headers: { 'Content-Type': 'application/json' },
   319→        body: JSON.stringify(body),
   320→      });
   321→
   322→      if (res.ok) {
   323→        const json = await res.json();
   324→        const conv: Conversation = json.data;
   325→        setConversations((prev) => [conv, ...prev]);
   326→        setNewConvOpen(false);
   327→        setNewConvParticipantId('');
   328→        setNewConvProjectId('');
   329→        openConversation(conv.id);
   330→        toast.success('مکالمه ایجاد شد');
   331→      } else {
   332→        const json = await res.json().catch(() => null);
   333→        toast.error(
   334→          json?.error?.message || 'خطا در ایجاد مکالمه',
   335→        );
   336→      }
   337→    } catch {
   338→      toast.error('خطا در ایجاد مکالمه');
   339→    } finally {
   340→      setCreatingConv(false);
   341→    }
   342→  }, [newConvParticipantId, newConvProjectId, creatingConv, openConversation]);
   343→
   344→  /* ---- Polling: messages ---- */
   345→  useEffect(() => {
   346→    if (!activeConversationId) return;
   347→    const interval = setInterval(() => {
   348→      if (lastFetchedMessageId.current) {
   349→        fetchMessages(activeConversationId, lastFetchedMessageId.current);
   350→      }
   351→    }, 5000);
   352→    return () => clearInterval(interval);
   353→  }, [activeConversationId, fetchMessages]);
   354→
   355→  /* ---- Polling: conversations (unread counts) ---- */
   356→  useEffect(() => {
   357→    const interval = setInterval(() => {
   358→      fetchConversations();
   359→    }, 10000);
   360→    return () => clearInterval(interval);
   361→  }, [fetchConversations]);
   362→
   363→  /* ---- Auto-scroll on new messages ---- */
   364→  useEffect(() => {
   365→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   366→  }, [messages]);
   367→
   368→  /* ---- Initial load ---- */
   369→  useEffect(() => {
   370→    fetchConversations();
   371→  }, [fetchConversations]);
   372→
   373→  /* ---- Back to list on mobile ---- */
   374→  const handleBackToList = useCallback(() => {
   375→    setShowChat(false);
   376→    // Don't clear activeConversationId so desktop keeps the chat open
   377→  }, []);
   378→
   379→  /* ---- Textarea auto-grow & keyboard handling ---- */
   380→  const handleTextareaChange = useCallback(
   381→    (e: React.ChangeEvent<HTMLTextAreaElement>) => {
   382→      const val = e.target.value;
   383→      if (val.length > 2000) return;
   384→      setMessageInput(val);
   385→
   386→      // Auto-grow
   387→      const ta = e.target;
   388→      ta.style.height = 'auto';
   389→      ta.style.height = Math.min(ta.scrollHeight, 120) + 'px';
   390→    },
   391→    [],
   392→  );
   393→
   394→  const handleKeyDown = useCallback(
   395→    (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   396→      if (e.key === 'Enter' && !e.shiftKey) {
   397→        e.preventDefault();
   398→        sendMessage();
   399→      }
   400→    },
   401→    [sendMessage],
   402→  );
   403→
   404→  /* ---- Filtered conversations ---- */
   405→  const filteredConversations = searchQuery.trim()
   406→    ? conversations.filter((c) => {
   407→        const other = getOtherParticipant(c, user?.id);
   408→        return other.displayName
   409→          .toLowerCase()
   410→          .includes(searchQuery.trim().toLowerCase());
   411→      })
   412→    : conversations;
   413→
   414→  /* ================================================================ */
   415→  /*  RENDER                                                           */
   416→  /* ================================================================ */
   417→
   418→  return (
   419→    <div className="mx-auto h-[calc(100vh-8rem)] max-w-6xl overflow-hidden rounded-xl border bg-surface shadow-sm">
   420→      <div className="flex h-full">
   421→        {/* ===================== LEFT PANEL ===================== */}
   422→        <div
   423→          className={cn(
   424→            'flex h-full w-full flex-col md:w-80 md:border-e',
   425→            showChat ? 'hidden md:flex' : 'flex',
   426→          )}
   427→        >
   428→          {/* Header */}
   429→          <div className="flex items-center justify-between border-b px-4 py-3">
   430→            <h1 className="text-lg font-bold">پیام‌ها</h1>
   431→
   432→            <Dialog open={newConvOpen} onOpenChange={setNewConvOpen}>
   433→              <DialogTrigger asChild>
   434→                <Button size="sm" variant="outline">
   435→                  <Plus className="ml-1 h-4 w-4" />
   436→                  مکالمه جدید
   437→                </Button>
   438→              </DialogTrigger>
   439→              <DialogContent>
   440→                <DialogHeader>
   441→                  <DialogTitle>مکالمه جدید</DialogTitle>
   442→                </DialogHeader>
   443→                <div className="mt-4 space-y-4">
   444→                  <div className="space-y-2">
   445→                    <label
   446→                      htmlFor="participantId"
   447→                      className="text-sm font-medium"
   448→                    >
   449→                      شناسه کاربر
   450→                    </label>
   451→                    <Input
   452→                      id="participantId"
   453→                      placeholder="شناسه کاربر مقابل را وارد کنید"
   454→                      value={newConvParticipantId}
   455→                      onChange={(e) => setNewConvParticipantId(e.target.value)}
   456→                    />
   457→                  </div>
   458→                  <div className="space-y-2">
   459→                    <label
   460→                      htmlFor="projectId"
   461→                      className="text-sm font-medium"
   462→                    >
   463→                      شناسه پروژه <span className="text-muted-foreground">(اختیاری)</span>
   464→                    </label>
   465→                    <Input
   466→                      id="projectId"
   467→                      placeholder="شناسه پروژه (در صورت وجود)"
   468→                      value={newConvProjectId}
   469→                      onChange={(e) => setNewConvProjectId(e.target.value)}
   470→                    />
   471→                  </div>
   472→                  <Button
   473→                    className="w-full"
   474→                    onClick={createConversation}
   475→                    disabled={!newConvParticipantId.trim() || creatingConv}
   476→                  >
   477→                    {creatingConv ? 'در حال ایجاد…' : 'ایجاد مکالمه'}
   478→                  </Button>
   479→                </div>
   480→              </DialogContent>
   481→            </Dialog>
   482→          </div>
   483→
   484→          {/* Search */}
   485→          <div className="border-b px-3 py-2">
   486→            <div className="relative">
   487→              <Search className="absolute start-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   488→              <Input
   489→                className="ps-9"
   490→                placeholder="جستجو در مکالمه‌ها…"
   491→                value={searchQuery}
   492→                onChange={(e) => setSearchQuery(e.target.value)}
   493→              />
   494→            </div>
   495→          </div>
   496→
   497→          {/* Conversation list */}
   498→          <ScrollArea className="flex-1">
   499→            {conversationsLoading ? (
   500→              <div className="space-y-3 p-3">
   501→                {Array.from({ length: 5 }).map((_, i) => (
   502→                  <div key={i} className="flex items-center gap-3 px-2 py-2">
   503→                    <Skeleton className="h-10 w-10 shrink-0 rounded-full" />
   504→                    <div className="flex-1 space-y-2">
   505→                      <Skeleton className="h-4 w-28" />
   506→                      <Skeleton className="h-3 w-40" />
   507→                    </div>
   508→                  </div>
   509→                ))}
   510→              </div>
   511→            ) : filteredConversations.length === 0 ? (
   512→              <div className="flex flex-col items-center justify-center px-4 py-16 text-center">
   513→                <div className="mb-3 flex h-14 w-14 items-center justify-center rounded-full bg-muted">
   514→                  <MessageCircle className="h-7 w-7 text-muted-foreground" />
   515→                </div>
   516→                <p className="text-sm text-muted-foreground">
   517→                  {searchQuery
   518→                    ? 'مکالمه‌ای یافت نشد'
   519→                    : 'هنوز مکالمه‌ای ندارید'}
   520→                </p>
   521→              </div>
   522→            ) : (
   523→              <div className="divide-y">
   524→                {filteredConversations.map((conv) => {
   525→                  const other = getOtherParticipant(conv, user?.id);
   526→                  const isActive = conv.id === activeConversationId;
   527→
   528→                  return (
   529→                    <button
   530→                      key={conv.id}
   531→                      className={cn(
   532→                        'flex w-full items-center gap-3 px-4 py-3 text-start transition-colors hover:bg-muted/50',
   533→                        isActive && 'border-s-2 border-primary bg-primary-soft/50',
   534→                      )}
   535→                      onClick={() => openConversation(conv.id)}
   536→                    >
   537→                      <Avatar className="h-10 w-10 shrink-0">
   538→                        {other.profile?.avatarUrl ? (
   539→                          <AvatarImage
   540→                            src={other.profile.avatarUrl}
   541→                            alt={other.displayName}
   542→                          />
   543→                        ) : null}
   544→                        <AvatarFallback className="text-xs">
   545→                          {getInitials(other.displayName)}
   546→                        </AvatarFallback>
   547→                      </Avatar>
   548→
   549→                      <div className="min-w-0 flex-1">
   550→                        <div className="flex items-center justify-between gap-2">
   551→                          <span className="truncate text-sm font-semibold">
   552→                            {other.displayName}
   553→                          </span>
   554→                          {conv.lastMessage && (
   555→                            <span className="shrink-0 text-xs text-muted-foreground">
   556→                              {relativeTime(conv.lastMessage.createdAt)}
   557→                            </span>
   558→                          )}
   559→                        </div>
   560→                        <div className="flex items-center justify-between gap-2">
   561→                          <p className="truncate text-xs text-muted-foreground">
   562→                            {conv.lastMessage
   563→                              ? truncate(conv.lastMessage.content, 50)
   564→                              : 'شروع گفتگو'}
   565→                          </p>
   566→                          {conv.unreadCount > 0 && (
   567→                            <Badge className="shrink-0 h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-xs">
   568→                              {formatNumber(conv.unreadCount)}
   569→                            </Badge>
   570→                          )}
   571→                        </div>
   572→                      </div>
   573→                    </button>
   574→                  );
   575→                })}
   576→              </div>
   577→            )}
   578→          </ScrollArea>
   579→        </div>
   580→
   581→        {/* ===================== RIGHT PANEL ==================== */}
   582→        <div
   583→          className={cn(
   584→            'flex h-full flex-1 flex-col',
   585→            !showChat ? 'hidden md:flex' : 'flex',
   586→          )}
   587→        >
   588→          {activeConversation ? (
   589→            <>
   590→              {/* Chat header */}
   591→              <div className="flex items-center gap-3 border-b px-4 py-3">
   592→                {/* Back button (mobile only) */}
   593→                <Button
   594→                  variant="ghost"
   595→                  size="icon"
   596→                  className="md:hidden"
   597→                  onClick={handleBackToList}
   598→                >
   599→                  <ArrowRight className="h-5 w-5" />
   600→                </Button>
   601→
   602→                <Avatar className="h-9 w-9 shrink-0">
   603→                  {(() => {
   604→                    const other = getOtherParticipant(
   605→                      activeConversation,
   606→                      user?.id,
   607→                    );
   608→                    return other.profile?.avatarUrl ? (
   609→                      <AvatarImage
   610→                        src={other.profile.avatarUrl}
   611→                        alt={other.displayName}
   612→                      />
   613→                    ) : null;
   614→                  })()}
   615→                  <AvatarFallback className="text-xs">
   616→                    {getInitials(
   617→                      getOtherParticipant(activeConversation, user?.id)
   618→                        .displayName,
   619→                    )}
   620→                  </AvatarFallback>
   621→                </Avatar>
   622→
   623→                <div className="min-w-0 flex-1">
   624→                  <p className="truncate text-sm font-semibold">
   625→                    {getOtherParticipant(activeConversation, user?.id)
   626→                      .displayName}
   627→                  </p>
   628→                  {activeConversation.type === 'PROJECT' && (
   629→                    <p className="text-xs text-muted-foreground">
   630→                      مکالمه پروژه‌ای
   631→                    </p>
   632→                  )}
   633→                </div>
   634→              </div>
   635→
   636→              {/* Messages area */}
   637→              <ScrollArea className="flex-1">
   638→                <div className="space-y-3 p-4">
   639→                  {messagesLoading ? (
   640→                    <div className="flex flex-col items-center gap-3 py-8">
   641→                      {Array.from({ length: 4 }).map((_, i) => (
   642→                        <Skeleton
   643→                          key={i}
   644→                          className={cn(
   645→                            'h-12 rounded-2xl',
   646→                            i % 2 === 0
   647→                              ? 'ms-auto w-3/5'
   648→                              : 'me-auto w-2/5',
   649→                          )}
   650→                        />
   651→                      ))}
   652→                    </div>
   653→                  ) : messages.length === 0 ? (
   654→                    <div className="flex items-center justify-center py-16">
   655→                      <p className="text-sm text-muted-foreground">
   656→                        هنوز پیامی ارسال نشده. اولین پیام را بفرستید!
   657→                      </p>
   658→                    </div>
   659→                  ) : (
   660→                    messages.map((msg) => {

   662→                      const isSystem = msg.type === 'SYSTEM';
   663→
   664→                      if (isSystem) {
   665→                        return (
   666→                          <div
   667→                            key={msg.id}
   668→                            className="flex justify-center"
   669→                          >
   670→                            <span className="rounded-full bg-muted px-3 py-1 text-xs text-muted-foreground">
   671→                              {msg.content}
   672→                            </span>
   673→                          </div>
   674→                        );
   675→                      }
   676→
   677→                      return (
   678→                        <div
   679→                          key={msg.id}
   680→                          className={cn(
   681→                            'flex flex-col',
   682→                            isMine ? 'items-start' : 'items-end',
   683→                          )}
   684→                        >
   685→                          {/* Sender name for PROJECT conversations */}
   686→                          {activeConversation.type === 'PROJECT' &&
   687→                            !isMine && (
   688→                              <span className="mb-1 text-xs text-muted-foreground">
   689→                                {(() => {
   690→                                  const sender = activeConversation.participants.find(
   691→                                    (p) => p.id === msg.senderId,
   692→                                  );
   693→                                  return sender?.displayName || '';
   694→                                })()}

   700→                              'rounded-2xl px-4 py-2 max-w-[75%]',

   706→                            <p className="whitespace-pre-wrap text-sm leading-7">
   707→                              {msg.content}
   708→                            </p>
   709→                          </div>
   710→                          <span
   711→                            className={cn(
   712→                              'mt-1 text-xs text-muted-foreground',
   713→                              isMine ? 'ms-1' : 'me-1',
   714→                            )}
   715→                          >
   716→                            {formatMessageTime(msg.createdAt)}
   717→                          </span>
   718→                        </div>
   719→                      );
   720→                    })
   721→                  )}
   722→                  <div ref={messagesEndRef} />
   723→                </div>
   724→              </ScrollArea>
   725→
   726→              {/* Message input */}
   727→              <div className="border-t p-3">
   728→                <div className="flex items-end gap-2">
   729→                  <textarea
   730→                    ref={textareaRef}
   731→                    value={messageInput}
   732→                    onChange={handleTextareaChange}
   733→                    onKeyDown={handleKeyDown}\n                    placeholder="پیام خود را بنویسید…"
   734→                    rows={1}
   735→                    className={cn(
   736→                      'flex-1 resize-none rounded-xl border bg-background px-4 py-2.5 text-sm',
   737→                      'focus:outline-none focus:ring-2 focus:ring-primary/30',
   738→                      'max-h-[120px] placeholder:text-muted-foreground',
   739→                    )}
   740→                  />
   741→                  <Button
   742→                    size="icon"
   743→                    className="h-10 w-10 shrink-0 rounded-xl"
   744→                    onClick={sendMessage}
   745→                    disabled={!messageInput.trim() || sendingMessage}
   746→                  >
   747→                    <PaperPlane className="h-4 w-4" />
   748→                  </Button>
   749→                </div>
   750→                {messageInput.length > 1800 && (
   751→                  <p className="mt-1 text-xs text-muted-foreground text-start">
   752→                    {formatNumber(2000 - messageInput.length)} کاراکتر باقی‌مانده
   753→                  </p>
   754→                )}
   755→              </div>
   756→            </>
   757→          ) : (
   758→            /* Empty state when no conversation selected */
   759→            <div className="flex h-full flex-col items-center justify-center px-4">
   760→              <div className="mb-4 flex h-20 w-20 items-center justify-center rounded-full bg-muted">
   761→                <MessageCircle className="h-10 w-10 text-muted-foreground" />
   762→              </div>
   763→              <h2 className="mb-2 text-lg font-semibold">پیام‌ها</h2>
   764→              <p className="max-w-xs text-center text-sm text-muted-foreground">
   765→                یک مکالمه را از لیست سمت راست انتخاب کنید یا مکالمه جدیدی بسازید.
   766→              </p>
   767→            </div>
   768→          )}
   769→        </div>
   770→      </div>
   771→    </div>
